cmake_minimum_required(VERSION 3.20)
project(uos_uapp_header_check CXX)
get_filename_component(UETD_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../UETD" ABSOLUTE)
set(kernel "${UETD_ROOT}/third_party/FreeRTOS-Kernel-11.3.1")
foreach(backend baremetal freertos)
  add_library(headers_${backend} OBJECT headers.cpp)
  target_include_directories(headers_${backend} PRIVATE "${UETD_ROOT}/include")
  target_compile_features(headers_${backend} PRIVATE cxx_std_17)
endforeach()
target_compile_definitions(headers_freertos PRIVATE UOS_FREERTOS_NATIVE_STORAGE=1)
target_include_directories(headers_freertos PRIVATE "${UETD_ROOT}/tests" "${kernel}/include" "${kernel}/portable/MSVC-MingW")
